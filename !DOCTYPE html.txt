      function drawRadar() {
        rCtx.clearRect(0, 0, radarCanvas.width, radarCanvas.height);
        const cx = radarCanvas.width / 2;
        const cy = radarCanvas.height / 2;

        // Background circles
        rCtx.strokeStyle = 'rgba(56, 189, 248, 0.15)';
        rCtx.lineWidth = 1;
        [35, 70, 105].forEach(radius => {
          rCtx.beginPath();
          rCtx.arc(cx, cy, radius, 0, Math.PI * 2);
          rCtx.stroke();
        });

        // Crosshairs
        rCtx.beginPath();
        rCtx.moveTo(cx, 10); rCtx.lineTo(cx, radarCanvas.height - 10);
        rCtx.moveTo(10, cy); rCtx.lineTo(radarCanvas.width - 10, cy);
        rCtx.stroke();

        // Rotating phosphor sweep beam
        rCtx.save();
        rCtx.translate(cx, cy);
        rCtx.rotate(radarAngle);
        const grad = rCtx.createLinearGradient(0, 0, 115, 0);
        grad.addColorStop(0, 'rgba(56, 189, 248, 0.6)');
        grad.addColorStop(1, 'rgba(56, 189, 248, 0)');
        rCtx.fillStyle = grad;
        rCtx.beginPath();
        rCtx.moveTo(0, 0);
        rCtx.arc(0, 0, 115, 0, -0.4, true);
        rCtx.fill();
        rCtx.restore();

        // Draw blips
        radarTargets.forEach(t => {
          const tx = cx + Math.cos(t.angle) * t.r;
          const ty = cy + Math.sin(t.angle) * t.r;
          rCtx.fillStyle = t.type === 'hostile' ? '#ef4444' : '#38bdf8';
          rCtx.shadowColor = rCtx.fillStyle;
          rCtx.shadowBlur = 6;
          rCtx.beginPath();
          rCtx.arc(tx, ty, 3.5, 0, Math.PI * 2);
          rCtx.fill();
        });

        radarAngle = (radarAngle + 0.035) % (Math.PI * 2);
        requestAnimationFrame(drawRadar);
      }

      function drawEcg() {
        ecgCtx.clearRect(0, 0, ecgCanvas.width, ecgCanvas.height);

        for (let x = 0; x < ecgCanvas.width; x++) {
          const cycle = (x + ecgOffset) % 130;
          let y = cy;
          if (cycle > 20 && cycle < 35) y -= Math.sin(((cycle - 20) / 15) * Math.PI) * 4;
          else if (cycle >= 45 && cycle < 48) y += 3;
          else if (cycle >= 48 && cycle < 56) y -= Math.sin(((cycle - 48) / 8) * Math.PI) * 24;
          else if (cycle >= 56 && cycle < 60) y += 5;
          else if (cycle >= 75 && cycle < 95) y -= Math.sin(((cycle - 75) / 20) * Math.PI) * 6;

          if (x === 0) ecgCtx.moveTo(x, y);
          else ecgCtx.lineTo(x, y);
        }
        ecgCtx.stroke();
        ecgOffset = (ecgOffset + 1.8) % 1300;
        requestAnimationFrame(drawEcg);

            // 3D Perspective Kinetic Impact Arena Simulation
      let kineticSparks = [];
      let screenShake = 0;

      function project3D(x, y, z) {
        const fov = 320;
        const pz = z + 360;
        const scale = fov / pz;
        return {
          x: simCanvas.width * 0.5 + x * scale,
          y: simCanvas.height * 0.5 + y * scale,
          scale
        };
      }

      function loopKineticSim() {
        sCtx.clearRect(0, 0, simCanvas.width, simCanvas.height);

        // Apply screen recoil shake on impact
        sCtx.save();
        if (screenShake > 0) {
          const sx = (Math.random() - 0.5) * screenShake;
          const sy = (Math.random() - 0.5) * screenShake;
          sCtx.translate(sx, sy);
          screenShake *= 0.88;
          if (screenShake < 0.2) screenShake = 0;
        }

        // Draw 3D Perspective Gridlines on floor
        sCtx.strokeStyle = 'rgba(56, 189, 248, 0.12)';
        sCtx.lineWidth = 1;
        for (let gz = 50; gz <= 350; gz += 50) {
          const p1 = project3D(-280, 80, gz);
          const p2 = project3D(280, 80, gz);
          sCtx.beginPath();
          sCtx.moveTo(p1.x, p1.y);
          sCtx.lineTo(p2.x, p2.y);
          sCtx.stroke();
        }
        for (let gx = -280; gx <= 280; gx += 70) {
          const p1 = project3D(gx, 80, 50);
          const p2 = project3D(gx, 80, 350);
          sCtx.beginPath();
          sCtx.moveTo(p1.x, p1.y);
          sCtx.lineTo(p2.x, p2.y);
          sCtx.stroke();
        }

        // 3D Shield Target at coordinate (110, -10, 80)
        const shieldCoord = { x: 110, y: -10, z: 80 };
        const sp = project3D(shieldCoord.x, shieldCoord.y, shieldCoord.z);

        // Draw 3D Angled Shield Disc
        sCtx.save();
        sCtx.translate(sp.x, sp.y);
        sCtx.scale(sp.scale, sp.scale * 0.88); // 3D oblique tilt

        // Outer red
        sCtx.fillStyle = '#ef4444';
        sCtx.beginPath();
        sCtx.arc(0, 0, 75, 0, Math.PI * 2);
        sCtx.fill();
        sCtx.strokeStyle = '#38bdf8';
        sCtx.lineWidth = 2;
        sCtx.stroke();

        // Silver
        sCtx.fillStyle = '#f8fafc';
        sCtx.beginPath();
        sCtx.arc(0, 0, 55, 0, Math.PI * 2);
        sCtx.fill();

        // Inner red
        sCtx.fillStyle = '#dc2626';
        sCtx.beginPath();
        sCtx.arc(0, 0, 38, 0, Math.PI * 2);
        sCtx.fill();

        // Blue center
        sCtx.fillStyle = '#0284c7';
        sCtx.beginPath();
        sCtx.arc(0, 0, 22, 0, Math.PI * 2);
        sCtx.fill();

        // Star
        sCtx.fillStyle = '#ffffff';
        sCtx.font = 'bold 16px sans-serif';
        sCtx.textAlign = 'center';
        sCtx.textBaseline = 'middle';
        sCtx.fillText('★', 0, 0);
        sCtx.restore();

        // 3D Projectile Trajectory in flight
        if (projectile.active) {
          projectile.progress += 0.058;
          const start = { x: -260, y: 55, z: 280 };
          const curX = start.x + (shieldCoord.x - start.x) * projectile.progress;
          const curY = start.y + (shieldCoord.y - start.y) * projectile.progress;
          const curZ = start.z + (shieldCoord.z - start.z) * projectile.progress;
          const curP = project3D(curX, curY, curZ);

          // 3D projectile glow
          sCtx.save();
          sCtx.fillStyle = '#ef4444';
          sCtx.shadowColor = '#ef4444';
          sCtx.shadowBlur = 15;
          sCtx.beginPath();
          sCtx.arc(curP.x, curP.y, 6 * curP.scale, 0, Math.PI * 2);
          sCtx.fill();

          // Plasma trail
          sCtx.strokeStyle = 'rgba(239, 68, 68, 0.7)';
          sCtx.lineWidth = 3.5 * curP.scale;
          const prevP = project3D(curX - 35, curY + 6, curZ + 25);
          sCtx.beginPath();
          sCtx.moveTo(prevP.x, prevP.y);
          sCtx.lineTo(curP.x, curP.y);
          sCtx.stroke();
          sCtx.restore();

          if (projectile.progress >= 1) {
            projectile.active = false;
            playSfx('clang');
            screenShake = 14;

            // Spawn 3D expanding shockwaves
            for (let i = 0; i < 4; i++) {
              waveRings.push({ x: sp.x, y: sp.y, r: 12 + i * 8, maxR: 180 + i * 25, alpha: 0.95, color: i % 2 === 0 ? '#38bdf8' : '#0284c7' });
            }

            // Spawn 80+ 3D spark particles
            for (let i = 0; i < 75; i++) {
              kineticSparks.push({
                x: shieldCoord.x,
                y: shieldCoord.y,
                z: shieldCoord.z,
                vx: (Math.random() - 0.7) * 14,
                vy: (Math.random() - 0.5) * 12,
                vz: (Math.random() - 0.2) * 12,
                alpha: 1,
                size: Math.random() * 2.5 + 1.2
              });
            }
          }
        }

        // Render expanding 3D shockwaves
        for (let i = waveRings.length - 1; i >= 0; i--) {
          const r = waveRings[i];
          r.r += 4.5;
          r.alpha -= 0.022;
          if (r.alpha <= 0 || r.r >= r.maxR) {
            waveRings.splice(i, 1);
            continue;
          }
          sCtx.save();
          sCtx.globalAlpha = Math.max(0, r.alpha);
          sCtx.strokeStyle = r.color;
          sCtx.lineWidth = 2.4;
          sCtx.shadowColor = r.color;
          sCtx.shadowBlur = 14;
          sCtx.beginPath();
          sCtx.ellipse(r.x, r.y, r.r, r.r * 0.75, 0, 0, Math.PI * 2);
          sCtx.stroke();
          sCtx.restore();
        }

        // Render 3D Sparks with gravity & perspective
        for (let i = kineticSparks.length - 1; i >= 0; i--) {
          const spk = kineticSparks[i];
          spk.x += spk.vx;
          spk.y += spk.vy;
          spk.z += spk.vz;
          spk.vy += 0.35; // gravity
          spk.alpha -= 0.024;

          if (spk.alpha <= 0) {
            kineticSparks.splice(i, 1);
            continue;
          }

          const pt = project3D(spk.x, spk.y, spk.z);
          sCtx.save();
          sCtx.globalAlpha = Math.max(0, spk.alpha);
          sCtx.fillStyle = spk.vz > 0 ? '#38bdf8' : '#ef4444';
          sCtx.beginPath();
          sCtx.arc(pt.x, pt.y, spk.size * pt.scale, 0, Math.PI * 2);
          sCtx.fill();
          sCtx.restore();
        }

        sCtx.restore();
        requestAnimationFrame(loopKineticSim);
      }
      loopKineticSim();
      loopKineticSim();

      // Glitch / Scramble Text Effect Utility
      function scrambleText(element, finalText, duration = 650) {
        const glyphs = '01!#$@%&*<>/\_?~ABCDEFGHIJKLMNOPQRSTUVWXYZ';
        const start = Date.now();
        const originalText = finalText;

        function update() {
          const elapsed = Date.now() - start;
          const progress = Math.min(1, elapsed / duration);
          const charsRevealed = Math.floor(progress * originalText.length);

          let output = originalText.substring(0, charsRevealed);
          for (let i = charsRevealed; i < originalText.length; i++) {
            if (originalText[i] === ' ') output += ' ';
            else output += glyphs[Math.floor(Math.random() * glyphs.length)];
          }
          element.textContent = output;

          if (progress < 1) requestAnimationFrame(update);
        }
        update();
      }

      function init3DCardTilt() {
        document.querySelectorAll('.hud-panel').forEach(card => {
          if (card.dataset.tiltInitialized) return;
          card.dataset.tiltInitialized = 'true';

          let glare = card.querySelector('.card-3d-glare');
          if (!glare) {
            glare = document.createElement('div');
            glare.className = 'card-3d-glare';
            card.appendChild(glare);
          }

          card.addEventListener('mousemove', (e) => {
            const rect = card.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            const cx = rect.width / 2;
            const cy = rect.height / 2;
            const rotX = -((y - cy) / cy) * 9.5;
            const rotY = ((x - cx) / cx) * 9.5;

            card.style.transform = `perspective(1000px) rotateX(${rotX.toFixed(2)}deg) rotateY(${rotY.toFixed(2)}deg) scale3d(1.02, 1.02, 1.02)`;
            glare.style.background = `radial-gradient(circle at ${((x / rect.width) * 100).toFixed(1)}% ${((y / rect.height) * 100).toFixed(1)}%, rgba(56, 189, 248, 0.28), transparent 60%)`;
            glare.style.opacity = '1';
          });

          card.addEventListener('mouseleave', () => {
            card.style.transform = 'perspective(1000px) rotateX(0deg) rotateY(0deg) scale3d(1, 1, 1)';
            glare.style.opacity = '0';
          });
        });
      }
      init3DCardTilt();

                scrollTrigger: {
                  trigger: sec,
                  start: 'top 85%',
                  toggleActions: 'play none none none'
                }
              }
            );
          }
        });
      }
